//! Compact port-spec parsing for a CLI: `"8080,443,9000-9005"` → the full port list.
//!
//! Application code, so errors use `anyhow` (`cargo add anyhow`).

use anyhow::{bail, ensure, Context, Result};

/// Expand a port spec like `"8080,443,9000-9005"` into every port it names,
/// with `lo-hi` ranges expanded inclusively. Spaces around commas and `-` are fine.
///
/// # Errors
/// Fails if the spec is empty, or any item is not a `u16` port or an
/// inclusive `lo-hi` range with `lo <= hi`.
pub fn parse_ports(spec: &str) -> Result<Vec<u16>> {
    if spec.trim().is_empty() {
        bail!("empty port spec");
    }
    let mut ports = Vec::new();
    for part in spec.split(',') {
        let part = part.trim();
        if let Some((lo, hi)) = part.split_once('-') {
            let lo: u16 = lo.trim().parse().with_context(|| format!("bad range start: {part:?}"))?;
            let hi: u16 = hi.trim().parse().with_context(|| format!("bad range end: {part:?}"))?;
            ensure!(lo <= hi, "invalid range: start {lo} > end {hi}");
            ports.extend(lo..=hi);
        } else {
            let port: u16 = part.parse().with_context(|| format!("invalid port: {part:?}"))?;
            ports.push(port);
        }
    }
    Ok(ports)
}
